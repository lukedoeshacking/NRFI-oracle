# Python packages this backend needs.
# Install them with: pip install -r requirements.txt

flask              # Web framework — used to build API routes
flask-cors         # Lets the frontend (browser) call this API without CORS errors
psycopg2-binary    # Driver that lets Python talk to a PostgreSQL database
xgboost            # Machine learning library (gradient boosting)
scikit-learn       # Machine learning tools (preprocessing, metrics, models)
pandas             # Tables and data analysis (DataFrames)
numpy              # Fast numeric arrays and math
requests           # Makes HTTP requests to other APIs
python-dotenv      # Loads secret config (like DATABASE_URL) from a .env file
joblib             # Save/load trained ML models to disk
gunicorn           # Production web server that runs the Flask app
